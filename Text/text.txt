#include <QCoreApplication>
#include "src/Monitor/Monitor.h"
#include <QSqlDatabase>
#include <QSqlError>
#include <QVariant>
#include <QSqlQuery>
#include <memory>
#include <QObject>
#include <QThread>
#include <QMutex>
#include <QVector>
#include <QtSnmp/QtSnmpPooler.h>
#include <QSqlDriver>

int main(int argc, char *argv[])
{
    QCoreApplication a(argc, argv);
    QMutex mtx;
    QSqlDatabase db = QSqlDatabase::addDatabase("QPSQL");
    db.setHostName("192.168.56.105");
    db.setDatabaseName("selection_monitoring");
    db.setUserName("postgres");
    db.setPassword("");
    db.setPort(5432);

    if(!db.open()){
        qDebug()<<db.lastError().text();
        return 1;
    }

    QSqlQuery query;
    if(!query.exec("SELECT e.id, e.element_name, e.ip_address, et.template_name, json_agg(json_build_object('id', pt.id, 'oid', pt.oid,'low_bound', th.low_bound, 'high_bound', th.high_bound)) AS params FROM elements e JOIN elements_template et ON e.element_template_id = et.id LEFT JOIN params_template pt ON et.id = pt.element_template_id LEFT JOIN thresholds th ON pt.threshold_id = th.id GROUP BY e.id, e.element_name, e.ip_address, et.template_name")){
        qDebug()<<db.lastError();
        return 1;
    }



    QList<BaseMonitor::MonitorInputData> inputData;



    while (query.next()) {
        BaseMonitor::MonitorInputData inputDataStruct;
        int element_id = query.value("id").toInt();
        QString ip_address = query.value("ip_address").toString();
        inputDataStruct.element_id = quint32(element_id);
        inputDataStruct.element_ip = ip_address;
        QJsonDocument jsonDoc = QJsonDocument::fromJson(query.value("params").toString().toUtf8());
        inputDataStruct.params = jsonDoc.array();
        inputData.append(inputDataStruct);


    }

    Monitor* m = new Monitor(&a, inputData);



    m->startMonitoring();


    qDebug()<<db.driver()->hasFeature(QSqlDriver::Transactions);

    QObject::connect(m, &Monitor::responseReceived,&a,[&db](BaseMonitor::monitoringItemDataResult midr){



      if (!db.transaction()){
          qDebug()<<"Transaction start failed"<<db.lastError().text();
          return;
      }
      QSqlQuery query;
      int element_integral_state = 1;
      query.prepare("INSERT INTO elements_params_state (element_id, value, param_template_id) VALUES (:element_id, :value, :param_template_id)");
      for(BaseMonitor::monitoringParamState& param_state: midr.params){


          if(param_state.param_bonds.low_bound!=0 && param_state.param_bonds.high_bound!=0){
              bool ok = true;
              int current_param_value = param_state.param_value.toInt(&ok);
              if(ok){
                  param_state.param_state =
                          current_param_value > param_state.param_bonds.low_bound
                          && current_param_value < param_state.param_bonds.high_bound;
              }
          }

          query.bindValue(":element_id",midr.element_id);
          query.bindValue(":value", param_state.param_value);
          query.bindValue(":param_template_id", param_state.param_template_id);

          if(!query.exec()){
              qDebug()<<"Error in inserting params_state"<<query.lastError().text();
              return;
          }
      }

      int errors = 0;
      for(const BaseMonitor::monitoringParamState& param_state: midr.params){
          if(!param_state.param_state){
              errors++;
          }
      }
      if(errors==0) element_integral_state = 1;
      else if(errors > 0 && errors<midr.params.size()) element_integral_state = 2;
      else element_integral_state = 3;

      query.clear();
      query.prepare("INSERT INTO elements_integral_state (element_id, integral_state) VALUES (:element_id, :integral_state) ON CONFLICT (element_id) DO UPDATE SET integral_state = EXCLUDED.integral_state");
      query.bindValue(":element_id", midr.element_id);
      query.bindValue(":integral_state", element_integral_state);

      if(!query.exec()){
          qDebug()<<"Error in evaluation state inserting "<<query.lastError();
          return;
      }
      if(!db.commit()){
          qDebug()<<"Commit error "<<db.lastError();
          db.rollback();
      }
      else{
          qDebug()<<"Transaction commited successfully";
      }


    });


    //db.close();

    return a.exec();
}
