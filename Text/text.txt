SELECT 
    e.id, 
    e.element_name, 
    e.ip, 
    et.template_name, 
    json_agg(
        json_build_object(
            'id', pt.id, 
            'oid', pt.oid, 
            'low_bound', th.low_bound, 
            'high_bound', th.high_bound
        )
    ) AS params
FROM elements e
JOIN element_template et ON e.element_template_id = et.id
LEFT JOIN params_template pt ON et.id = pt.element_template_id
LEFT JOIN thresholds th ON pt.threshold_id = th.id
GROUP BY e.id, e.element_name, e.ip, et.template_name;
